cmake_minimum_required(VERSION 3.25)
project(octopus_metrics VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(METRICS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(METRICS_TSAN "Enable thread sanitizer (separate build)" OFF)
if(METRICS_SANITIZE AND METRICS_TSAN)
  message(FATAL_ERROR "Select only one sanitizer configuration")
endif()
find_package(Threads REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(CURL 7.85 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis>=1.2)
pkg_check_modules(SIMDJSON REQUIRED IMPORTED_TARGET simdjson>=3.0)

# The binary is tied to the canonical schema it was built against, never a
# caller-supplied checksum that could bypass fail-closed preflight.
set(MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/../../sql/postgres/octopus_core/manifest.yaml")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MANIFEST}")
file(READ "${MANIFEST}" SCHEMA)
string(REGEX MATCH "manifest_sha256: ([0-9a-f]+)" _ "${SCHEMA}")
set(MANIFEST_SHA "${CMAKE_MATCH_1}")
string(LENGTH "${MANIFEST_SHA}" HASH_LENGTH)
if(NOT HASH_LENGTH EQUAL 64)
  message(FATAL_ERROR "Missing canonical manifest checksum")
endif()
string(REGEX MATCH "schema_version: ([0-9]+)" _ "${SCHEMA}")
set(MANIFEST_VERSION "${CMAKE_MATCH_1}")
if(MANIFEST_VERSION STREQUAL "")
  message(FATAL_ERROR "Missing canonical manifest schema version")
endif()
add_library(metrics_core src/domain.cpp src/config.cpp src/adapters.cpp src/service.cpp)
target_include_directories(metrics_core PUBLIC include)
target_link_libraries(metrics_core PUBLIC Threads::Threads PostgreSQL::PostgreSQL CURL::libcurl
  PkgConfig::HIREDIS PkgConfig::SIMDJSON)
target_compile_definitions(metrics_core PRIVATE
  METRICS_MANIFEST_SHA="${MANIFEST_SHA}" METRICS_MANIFEST_VERSION="${MANIFEST_VERSION}")
target_compile_options(metrics_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(METRICS_SANITIZE)
  target_compile_options(metrics_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(metrics_core PUBLIC -fsanitize=address,undefined)
elseif(METRICS_TSAN)
  target_compile_options(metrics_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(metrics_core PUBLIC -fsanitize=thread)
endif()
add_executable(octopus-metrics src/main.cpp)
target_link_libraries(octopus-metrics PRIVATE metrics_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(metrics-tests tests/core_test.cpp)
  target_link_libraries(metrics-tests PRIVATE metrics_core)
  add_test(NAME metrics-core COMMAND metrics-tests)
  set_tests_properties(metrics-core PROPERTIES TIMEOUT 30)
  add_executable(metrics-probe tests/adapter_probe.cpp)
  target_link_libraries(metrics-probe PRIVATE metrics_core)
endif()
install(TARGETS octopus-metrics RUNTIME DESTINATION bin)
